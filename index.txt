    // ห่อด้วย DOMContentLoaded เพื่อให้มั่นใจว่าโหลด HTML เสร็จก่อนสร้างกราฟ
    document.addEventListener("DOMContentLoaded", function () {
      const canvas = document.getElementById('healthChart');
      if (!canvas) return;

      const ctx = canvas.getContext('2d');
      
      const labels = ['7 ก.ย. (จ)', '8 ก.ย. (อ)', '9 ก.ย. (พ)', '10 ก.ย. (พฤ)', '11 ก.ย. (ศ)', '12 ก.ย. (ส)', '13 ก.ย. (อา)'];

      const waterData = [1.8, 1.5, 1.5, 1.5, 1.5, 1.5, 1.5];

      // ตรวจสอบว่า Chart loaded หรือไม่
      if (typeof Chart === 'undefined') {
        canvas.parentNode.innerHTML = '<p style="color:red; text-align:center; padding: 20px;">ไม่สามารถโหลดไลบรารีกราฟได้ กรุณาตรวจสอบการเชื่อมต่ออินเทอร์เน็ต</p>';
        return;
      }

      new Chart(ctx, {
        type: 'bar',
        data: {
          labels: labels,
          datasets: [
            {
              label: 'ปริมาณน้ำ (ลิตร)',
              data: waterData,
              backgroundColor: '#0284c7',
              borderRadius: 6,
              yAxisID: 'yWater',
              order: 2
            },
            {
              label: 'เวลานอน (ชั่วโมง)',
              data: sleepData,
              borderColor: '#8b5cf6',
              backgroundColor: '#8b5cf6',
              type: 'line',
              tension: 0.2,
              borderWidth: 3,
              pointRadius: 4,
              yAxisID: 'ySleep',
              order: 1
            }
          ]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          plugins: {
            legend: {
              position: 'bottom',
              labels: { font: { family: 'Sarabun' } }
            },
            tooltip: {
              mode: 'index',
              intersect: false,
              bodyFont: { family: 'Sarabun' }
            }
          },
          scales: {
            x: {
              grid: { display: false },
              ticks: { font: { family: 'Sarabun' } }
            },
            ySleep: {
              type: 'linear',
              position: 'left',
              min: 0,
              max: 10,
              title: { display: true, text: 'ชั่วโมงนอน (ชม.)', font: { family: 'Sarabun' } },
              grid: { borderDash: [2, 4] }
            },
            yWater: {
              type: 'linear',
              position: 'right',
              min: 0,
              max: 3,
              title: { display: true, text: 'น้ำดื่ม (ลิตร)', font: { family: 'Sarabun' } },
              grid: { display: false }
            }
          }
        }
      });
    });
